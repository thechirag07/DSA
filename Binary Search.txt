//Binary Search
#include <stdio.h>

int main()
{
    int i,j,n,key, a[10];
    int low, high, m, found=0;

    printf("\n************* Program for Binary Search *************");
    printf("\n \n Number of element in an array n=:");
    scanf("\n %d", &n);

    printf("\n Element in an array: ");
    for(i=0;i<n;i++)
    {
        scanf("\t %d", &a[i]);
    }

    printf("\n \n key element to search in an array n=:");
    scanf("\n %d", &key);

    //Display Original Array
    printf("\n Original array is: ");
    for(i=0;i<n;i++)
    {
        printf("\n \n a[%d]= %d", i, a[i]);
    }

    // Binary Search main Code
    low=0;high=n-1;

    while(low<=high)
    {
        m=(low+high)/2;

        if(a[m]==key)
        {
            found=1;
            break;
        }
        else if(key<a[m])
            high=m-1;
        else
            low =m+1;
    }

    if(found==1)
        printf("\n\n \n key=%d is present at a[%d] location", key, m);
    else
        printf("\n Key=%d is not present in given array", key);

    return 0;
}
